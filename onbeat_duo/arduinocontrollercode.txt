const int buttonPins[] = {8, 9, 10, 11, 12};
const int ledPins[] = {3, 4, 5, 6, 7};
const int numButtons = 5;

// Instead of one variable, we track the last state of every button
// Using an array allows us to detect simultaneous changes
int lastButtonStates[numButtons]; 

void setup() {
  Serial.begin(9600);

  for (int i = 0; i < numButtons; i++) {
    pinMode(buttonPins[i], INPUT_PULLUP);
    pinMode(ledPins[i], OUTPUT);
    digitalWrite(ledPins[i], LOW);
    
    // Initialize all buttons as "not pressed" (HIGH because of INPUT_PULLUP)
    lastButtonStates[i] = HIGH; 
  }
}

void loop() {
  for (int i = 0; i < numButtons; i++) {
    // 1. Read the current state of this specific button
    int currentState = digitalRead(buttonPins[i]);

    // 2. Check if the state changed from the last time we looked
    if (currentState != lastButtonStates[i]) {
      
      // 3. If it changed to LOW, that's a "New Press"
      if (currentState == LOW) {
        digitalWrite(ledPins[i], HIGH);
        Serial.println(i + 1); // Send ONLY the button that just changed
      } 
      // 4. If it changed back to HIGH, the button was released
      else {
        digitalWrite(ledPins[i], LOW);
      }

      // 5. Save the state so we don't spam the serial port
      lastButtonStates[i] = currentState;
    }
  }

  delay(10); // Slightly faster debounce for rhythm gaming
}
